# NeuroBPF Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a portfolio-grade intrusion-detection pipeline: eBPF kernel capture + synthetic scenario engine → temporal provenance graph → reconstruction-anomaly GNN → live 2D/3D visualizer.

**Architecture:** A shared canonical NDJSON event schema feeds one graph builder (sliding temporal windows producing provenance snapshots). A graph-autoencoder GNN trained only on normal-window snapshots scores every node by reconstruction error; flagged nodes highlight in a React/force-graph visualizer. A C+libbpf collector emits the same schema from the real kernel.

**Tech Stack:** Python 3.14 + PyTorch 2.14 (CPU) + pytest; Node 26 + Vite + React + react-force-graph; C + libbpf + clang + Makefile. Host: CachyOS, kernel 7.2, BTF present.

**Spec:** `docs/superpowers/specs/2026-09-13-neurobpf-design.md`

```python
from __future__ import annotations

import json
from dataclasses import asdict, dataclass
from pathlib import Path
from typing import Iterable, Iterator


@dataclass
class Event:
    ts: float
    event: str
    pid: int
    ppid: int = -1
    comm: str = ""
    exe_path: str = ""
    uid: int = 1000
    target_type: str | None = None
    target: str | int | None = None
    result: int = 0

    def to_dict(self) -> dict:
        d = asdict(self)
        if d["target_type"] is None:
            d["target_type"] = None
        return d


def event_to_json(e: Event) -> str:
    return json.dumps(e.to_dict(), sort_keys=True, separators=(",", ":"))


def event_from_json(line: str) -> Event:
    return Event(**json.loads(line))


def write_events(path: Path, events: Iterable[Event]) -> None:
    with open(path, "w") as f:
        for e in events:
            f.write(event_to_json(e) + "\n")


def read_events(path: Path) -> Iterator[Event]:
    with open(path) as f:
        for line in f:
            line = line.strip()
            if not line:
                continue
            yield event_from_json(line)
```

from typing import Iterable

        self.nodes = nodes
        self.edges = edges
        self.malicious = set(malicious)

def build_snapshots(events, window=5.0, malicious_pids=()):
    malicious_pids = set(malicious_pids)
    registry = {}
    dirty_pids = set()
    edges = []
    touches = defaultdict(set)
    snapshots = []
    start_ts = None
    emitted = 0

    for ev in events:
        if start_ts is None:
            start_ts = ev.ts
        next_win = start_ts + window
        if ev.ts >= next_win:
            snapshots.append(_snapshot_from(registry, edges, touches, malicious_pids, start_ts))
            emitted += 1
            start_ts = ev.ts

        if ev.event == "process_start":
            nid = f"p:{ev.pid}"
            registry[nid] = NodeData(nid, "process", ev.comm or ev.exe_path or str(ev.pid), uid=ev.uid, pid=ev.pid)
            if ev.ppid > 0:
                parent = f"p:{ev.ppid}"
                if parent in registry:
                    edges.append(EdgeData(parent, nid, "EXEC", ev.ts))
            dirty_pids.add(ev.pid)
        elif ev.event == "process_exit":
            dirty_pids.add(ev.pid)
        elif ev.event in ("file_read", "file_write", "file_unlink"):
            nid = _file_node(registry, ev, dirty_pids, touches)
            label = "READ" if ev.event == "file_read" else "WRITE" if ev.event == "file_write" else "UNLINK"
            edges.append(EdgeData(f"p:{ev.pid}", nid, label, ev.ts))
            _touch(ev.pid, nid, touches)
        elif ev.event == "file_rename":
            old = _file_node(registry, ev.with_target(), dirty_pids, touches)
            ...
        elif ev.event in ("net_connect", "net_accept"):
            nid = _net_node(registry, ev, dirty_pids, touches)
            label = "CONNECT" if ev.event == "net_connect" else "ACCEPT"
            edges.append(EdgeData(f"p:{ev.pid}", nid, label, ev.ts))
            _touch(ev.pid, nid, touches)

    if emitted == 0 or True:
        snapshots.append(_snapshot_from(registry, edges, touches, malicious_pids, start_ts or 0.0))
    return snapshots

Common helpers inside `scenarios.py`:
- `_shell_chain(sim, attacker_pid, payload_command)` — spawn `sh -c`, write/exec payload.
- Each scenario deterministically consumes `PidSource` pids; attacker is a distinct pid lineage (e.g., spawned from a user process or systemd) so its graph branch is structurally novel.

Chains (all timed within one or two 5 s windows to land in few snapshots):
1. `code_injection`: background process (e.g., a server `app` pid) gets "hijacked" → forks `sh` → `sh` writes `/tmp/.payload` (mode 0777) → execs `/tmp/.payload` → payload does `net_connect` to `IP:1337`. Malicious = all involved pids.
2. `persistence`: attacker shells write `~/.bashrc`, `/etc/crontab`, `/.config/systemd/user/p.service` and drop `/tmp/.bin` (0777) + write+exec it at window end.
3. `exfiltration`: normal baseline, then a user process `svc` starts rapid `net_connect` burst (5+ connects within 2 s) to random high ports 40000–65535 on `10.9.9.x`; also a slow DNS to `dns.evil.example`. Malicious = `svc`.
4. `lateral_movement`: a `sshd`-like pid receives `net_accept` from `10.1.1.7:22`, then an `scp`-like child runs (fork/exec), writes `~/.ssh/authorized_keys`, reads `/etc/shadow`. Malicious = sshd + children.
5. `supply_chain`: `pkgmgr` connects to `registry.evil.example:443`, `file_write` of `/tmp/pkg.tar` + extract, then execs `/tmp/pkg/bin/pkg` which connects to `10.5.5.5:9000` and writes `.cache`. Malicious = pkgmgr + exec'd pkg.

Each returns `ScenarioResult(events, malicious_pids, name, run_type="attack")`.

```python
from __future__ import annotations

import torch

from ..graphbuilder import Snapshot
from ..events import ...

EDGE_TYPES = ("EXEC", "READ", "WRITE", "UNLINK", "CONNECT", "ACCEPT")
FEAT_DIM = 12
_KIND_DIM = 3
_ETYPE_DIM = len(EDGE_TYPES)


@dataclass
class GraphTensor:
    x: torch.Tensor
    edge_index: torch.LongTensor
    edge_type: torch.LongTensor
    node_ids: list[str]
    malicious: list[bool]
    snapshot_id: int


def snapshot_to_graph(snap, snapshot_id=0):
    ids = list(snap.nodes.keys())
    index = {nid: i for i, nid in enumerate(ids)}
    n = len(ids)
    adj_in = {nid: [] for nid in ids}
    adj_out = {nid: [] for nid in ids}
    e_src, e_dst, e_type = [], [], []
    for e in snap.edges:
        if e.src in index and e.dst in index:
            e_src.append(index[e.src]); e_dst.append(index[e.dst])
            e_type.append(EDGE_TYPES.index(e.label))
            adj_out[e.src].append(e.dst)
            adj_in[e.dst].append(e.src)
    max_deg = max((len(adj_out[nid] + adj_in[nid]) for nid in ids), default=1)
    feats = []
    for nid in ids:
        nd = snap.nodes[nid]
        kind = [float(nid.startswith(f"{k}:")) for k in ("p", "f", "n")]
        is_root = float(nd.uid == 0)
        deg_in = len(adj_in[nid]) / max_deg
        deg_out = len(adj_out[nid]) / max_deg
        evcount = _count_edges(snap, nid)
        feats.append(kind + [is_root, deg_in, deg_out] + evcount)
    x = torch.tensor(feats, dtype=torch.float32)
    edge_index = torch.tensor([e_src, e_dst], dtype=torch.long)
    edge_type = torch.tensor(e_type, dtype=torch.long)
    malicious = [nid in snap.malicious for nid in ids]
    return GraphTensor(x=x, edge_index=edge_index, edge_type=edge_type,
                       node_ids=ids, malicious=malicious, snapshot_id=snapshot_id)


def _count_edges(snap, nid):
    counts = [0] * _ETYPE_DIM
    for e in snap.edges:
        if e.src == nid or e.dst == nid:
            counts[EDGE_TYPES.index(e.label)] += 1
    mx = max(counts) if max(counts) else 1
    return [c / mx for c in counts]
```

```python
from __future__ import annotations

import torch
import torch.nn as nn
import torch.nn.functional as F


class GCNConv(nn.Module):
    def __init__(self, in_dim, out_dim, bias=True):
        super().__init__()
        self.weight = nn.Parameter(torch.empty(in_dim, out_dim))
        if bias:
            self.bias = nn.Parameter(torch.zeros(out_dim))
        else:
            self.register_parameter("bias", None)
        nn.init.xavier_uniform_(self.weight)

    def forward(self, x, adj):
        n = x.shape[0]
        a = adj + torch.eye(n, device=adj.device)
        deg = a.sum(dim=1).clamp(min=1.0)
        dinv = (1.0 / torch.sqrt(deg))[:, None]
        norm = a * dinv
        norm = norm * dinv.T
        out = norm @ x @ self.weight
        if self.bias is not None:
            out = out + self.bias
        return out


class GAE(nn.Module):
    def __init__(self, in_dim, hidden_dim=64, z_dim=64, dropout=0.3, pos_weight=8.0):
        super().__init__()
        self.conv1 = GCNConv(in_dim, hidden_dim)
        self.conv2 = GCNConv(hidden_dim, z_dim)
        self.dropout = nn.Dropout(dropout)
        self.activation = nn.ReLU()
        self.pos_weight = torch.tensor(pos_weight)

    def encode(self, x, adj):
        z = self.activation(self.conv1(x, adj))
        z = self.dropout(z)
        z = self.conv2(z, adj)
        return z

    def forward(self, x, adj):
        return self.encode(x, adj)

    def decode(self, z):
        return z @ z.t()

    def reconstruct_logits(self, x, adj):
        return self.decode(self.encode(x, adj))

    def train_loss(self, x, adj):
        logits = self.reconstruct_logits(x, adj)
        pw = torch.ones_like(adj) if not (adj > 0).any() else self.pos_weight.to(x.device)
        return F.binary_cross_entropy_with_logits(logits, adj, pos_weight=torch.where((adj > 0), self.pos_weight.to(x.device), torch.ones_like(adj)))

    def node_scores(self, x, adj):
        logits = self.reconstruct_logits(x, adj)
        probs = torch.sigmoid(logits)
        err = F.binary_cross_entropy(probs, adj, reduction="none")
        n = adj.shape[0]
        norm = err + err.t()
        return norm.mean(dim=1)
```

`train_loss`: use `pos_weight` tensor per element (pos edges weight 8.0). This yields mean BCE. `node_scores`: symmetrized per-node mean BCE.
